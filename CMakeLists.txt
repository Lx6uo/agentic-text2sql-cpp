cmake_minimum_required(VERSION 3.20)
project(agentic_text2sql VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TEXT2SQL_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

# Bundle data is local: clone + build needs no extra dataset download or Python.
set(TEXT2SQL_DATA_DIR "${CMAKE_CURRENT_BINARY_DIR}/data")
file(MAKE_DIRECTORY "${TEXT2SQL_DATA_DIR}")
configure_file(data/college_2.sqlite "${TEXT2SQL_DATA_DIR}/college_2.sqlite" COPYONLY)
configure_file(data/company.sqlite "${TEXT2SQL_DATA_DIR}/company.sqlite" COPYONLY)
file(COPY data/sample-spider DESTINATION "${TEXT2SQL_DATA_DIR}")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/data/spider.zip" TEXT2SQL_ARCHIVE_SHA)
set(TEXT2SQL_DATA_STAMP "${TEXT2SQL_DATA_DIR}/spider/.bundle-sha256")
set(TEXT2SQL_PREVIOUS_SHA "")
if(EXISTS "${TEXT2SQL_DATA_STAMP}")
  file(READ "${TEXT2SQL_DATA_STAMP}" TEXT2SQL_PREVIOUS_SHA)
endif()
if(NOT TEXT2SQL_PREVIOUS_SHA STREQUAL TEXT2SQL_ARCHIVE_SHA)
  message(STATUS "Extracting bundled Spider dataset")
  file(REMOVE_RECURSE "${TEXT2SQL_DATA_DIR}/spider")
  file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/data/spider.zip"
       DESTINATION "${TEXT2SQL_DATA_DIR}")
  file(WRITE "${TEXT2SQL_DATA_STAMP}" "${TEXT2SQL_ARCHIVE_SHA}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS data/spider.zip)
if(MSVC AND VCPKG_TARGET_TRIPLET MATCHES "-windows-static$")
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(CURL 7.68 REQUIRED)
add_library(sqlite STATIC third_party/sqlite/sqlite3.c)
target_include_directories(sqlite PUBLIC third_party/sqlite)
target_compile_definitions(sqlite PRIVATE SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0 SQLITE_DEFAULT_MEMSTATUS=0)
find_package(Threads REQUIRED)
target_link_libraries(sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_library(cjson STATIC third_party/cjson/cJSON.c)
target_include_directories(cjson PUBLIC third_party/cjson)
target_compile_definitions(cjson PUBLIC CJSON_HIDE_SYMBOLS)
if(NOT WIN32)
  target_link_libraries(cjson PUBLIC m)
endif()
if(TEXT2SQL_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "TEXT2SQL_SANITIZERS currently requires Clang or GCC")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(sqlite PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(cjson PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
# main.cpp includes the CPP modules; compiling them again would duplicate definitions.
add_executable(text2sql src/main.cpp)
target_compile_definitions(text2sql PRIVATE TEXT2SQL_DATA_DIR="${TEXT2SQL_DATA_DIR}")
target_link_libraries(text2sql PRIVATE sqlite cjson CURL::libcurl)
if(WIN32)
  target_link_libraries(text2sql PRIVATE shell32)
endif()
if(MSVC)
  target_compile_options(text2sql PRIVATE /W4 /permissive- /utf-8)
  target_compile_definitions(text2sql PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(text2sql PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS text2sql RUNTIME DESTINATION bin)
install(DIRECTORY "${TEXT2SQL_DATA_DIR}/" DESTINATION share/text2sql/data)
install(FILES config.example.json DESTINATION share/text2sql)
