cmake_minimum_required(VERSION 3.20)
project(agentic_text2sql VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TEXT2SQL_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(MSVC AND VCPKG_TARGET_TRIPLET MATCHES "-windows-static$")
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(CURL 7.68 REQUIRED)
add_library(sqlite STATIC third_party/sqlite/sqlite3.c)
target_include_directories(sqlite PUBLIC third_party/sqlite)
target_compile_definitions(sqlite PRIVATE SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0 SQLITE_DEFAULT_MEMSTATUS=0)
find_package(Threads REQUIRED)
target_link_libraries(sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_library(text2sql_core src/database.cpp src/config.cpp src/llm.cpp src/agent.cpp src/benchmark.cpp)
target_include_directories(text2sql_core PUBLIC src third_party/json)
target_link_libraries(text2sql_core PUBLIC sqlite CURL::libcurl)
if(MSVC)
  target_compile_options(text2sql_core PRIVATE /W4 /permissive- /utf-8)
  target_compile_definitions(text2sql_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(text2sql_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(TEXT2SQL_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "TEXT2SQL_SANITIZERS currently requires Clang or GCC")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(sqlite PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(text2sql_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
enable_testing()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/examples/demo.sql" DEMO_SQL)
configure_file(src/demo.h.in generated/demo.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS examples/demo.sql)
add_executable(text2sql src/main.cpp)
target_include_directories(text2sql PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(text2sql PRIVATE text2sql_core)
if(WIN32)
  target_link_libraries(text2sql PRIVATE shell32)
endif()
if(MSVC)
  target_compile_options(text2sql PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(text2sql PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS text2sql RUNTIME DESTINATION bin)
add_executable(text2sql_tests tests/tests.cpp)
target_link_libraries(text2sql_tests PRIVATE text2sql_core)
add_executable(benchmark_tests tests/benchmark_tests.cpp)
target_link_libraries(benchmark_tests PRIVATE text2sql_core)
if(MSVC)
  target_compile_options(text2sql_tests PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(benchmark_tests PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(text2sql_tests PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(benchmark_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME core COMMAND text2sql_tests)
add_test(NAME benchmark COMMAND benchmark_tests)
add_test(NAME cli_help COMMAND text2sql --help)
add_test(NAME cli_demo COMMAND text2sql demo --repair --json)
add_test(NAME cli_workflow COMMAND ${CMAKE_COMMAND}
  -DTEXT2SQL=$<TARGET_FILE:text2sql>
  -DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}/cli-test-workspace
  -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.cmake)
set_tests_properties(core benchmark PROPERTIES TIMEOUT 30)
set_tests_properties(cli_help cli_demo cli_workflow PROPERTIES TIMEOUT 30)
